# Test-only plugins, built next to the example in <build>/plugins.
find_package(Threads REQUIRED)
clpd_plugin_module(test_hooks_a hooks.c)
target_compile_definitions(test_hooks_a PRIVATE HOOKS_NAME="hooks-a")
clpd_plugin_module(test_hooks_b hooks.c)
target_compile_definitions(test_hooks_b PRIVATE HOOKS_NAME="hooks-b")
clpd_plugin_module(test_fail_init hooks.c)
target_compile_definitions(test_fail_init PRIVATE HOOKS_NAME="fail-init" HOOKS_FAIL_INIT)
clpd_plugin_module(test_big hooks.c)
target_compile_definitions(test_big PRIVATE HOOKS_NAME="big" HOOKS_BIG)
# The longest valid name: 64 characters.
string(REPEAT "a" 64 name64)
clpd_plugin_module(test_name64 hooks.c)
target_compile_definitions(test_name64 PRIVATE HOOKS_NAME="${name64}")
clpd_plugin_module(test_thread hooks.c)
target_compile_definitions(test_thread PRIVATE HOOKS_NAME="thread" HOOKS_THREAD)
target_link_libraries(test_thread PRIVATE Threads::Threads)
clpd_plugin_module(test_markers hooks.c)
target_compile_definitions(test_markers PRIVATE HOOKS_NAME="markers" HOOKS_MARKERS)
foreach(defect nosym abi size size1 null name name65 noinit undef internal)
  string(TOUPPER ${defect} upper)
  clpd_plugin_module(test_bad_${defect} bad.c)
  target_compile_definitions(test_bad_${defect} PRIVATE BAD_${upper})
endforeach()

# Prints sizeof(clpd_plugin) for the CLI test, which cannot compute it.
add_executable(plugin_desc_size desc_size.c)
target_link_libraries(plugin_desc_size PRIVATE clpd_plugin_api)
target_compile_options(plugin_desc_size PRIVATE ${CLPD_WARNINGS})
